-- Makine listesi: tonajlar, makine seçimleri, joker işaretleri ve grup sayısı burada saklanır. -- Supabase > SQL Editor'de bir kez çalıştırın. create table if not exists public.makine_listesi ( id int primary key default 1 check (id = 1), deger jsonb not null default '{}'::jsonb, guncelleme timestamptz not null default now() ); alter table public.makine_listesi enable row level security; -- Yönetici mi? (profiles.rol = 'admin' ve yasaklı değil) create or replace function public.makine_admin_mi() returns boolean language sql stable security definer set search_path = public as $$ select exists ( select 1 from public.profiles p where p.id = auth.uid() and p.rol = 'admin' and coalesce(p.banli, false) = false ); $$; drop policy if exists makine_listesi_oku on public.makine_listesi; create policy makine_listesi_oku on public.makine_listesi for select to anon, authenticated using (true); drop policy if exists makine_listesi_ekle on public.makine_listesi; create policy makine_listesi_ekle on public.makine_listesi for insert to authenticated with check (public.makine_admin_mi()); drop policy if exists makine_listesi_guncelle on public.makine_listesi; create policy makine_listesi_guncelle on public.makine_listesi for update to authenticated using (public.makine_admin_mi()) with check (public.makine_admin_mi()); grant select on public.makine_listesi to anon, authenticated; grant insert, update on public.makine_listesi to authenticated; -- Canlı güncelleme (yönetici kaydedince diğer cihazlar anında görsün) do $$ begin alter publication supabase_realtime add table public.makine_listesi; exception when duplicate_object then null; end $$;
